\documentclass[tikz, border=0]{standalone}
\usepackage{amsmath}
\usetikzlibrary{calc,arrows.meta,patterns, shapes.geometric, calc}

\definecolor{Red}{RGB}{ 159, 36, 48 }
\definecolor{Blue}{RGB}{ 103, 178, 255 }
\definecolor{LBlue}{RGB}{ 186, 220, 255 }
\definecolor{DBlue}{HTML}{132C53} 
\definecolor{gray}{HTML}{708090}
\definecolor{orange}{HTML}{ed9552}
\definecolor{orange2}{HTML}{e29d40}


\NewDocumentCommand{\DrawPerson}{O{(0,0)} O{DBlue} O{white} O{1}}{

\begin{scope}[shift={#1}]

\coordinate (head) at (0,{3.5*\Head});

\coordinate (fixed-joint-sholders) at (0,{2.5*\Head});
\coordinate (fixed-joint-hips) at (0,0);
\coordinate (fixed-joint-head) at (0,{3*\Head});

\coordinate (joint-left-sholder) at ($ (fixed-joint-sholders) + ({SholderLx(0)}, {SholderLy(0)})$);
\coordinate (joint-right-sholder) at ($ (fixed-joint-sholders) + ({SholderRx(0)}, {SholderRy(0)})$);

\coordinate (joint-left-elbow) at ($ (joint-left-sholder) + (270-\elbowAngle:{1.5*\Head}) $);
\coordinate (joint-right-elbow) at ($ (joint-right-sholder) + (-90+\elbowAngle:{1.5*\Head}) $);

\coordinate (joint-left-wrist) at ($ (joint-left-elbow) + (270-\elbowAngle-\wristAngle:{\Head}) $);
\coordinate (joint-right-wrist) at ($ (joint-right-elbow) + (-90+\elbowAngle+\wristAngle:{\Head}) $);

\coordinate (joint-left-hip) at ($ (fixed-joint-hips) + ({HipsLx(0)}, {HipsLy(0)})$);
\coordinate (joint-right-hip) at ($ (fixed-joint-hips) + ({HipsRx(0)}, {HipsRy(0)})$);

\coordinate (joint-left-knee) at ($ (joint-left-hip) + (270-\kneeAngle:{2*\Head})$);
\coordinate (joint-right-knee) at ($ (joint-right-hip) + (-90+\kneeAngle:{2*\Head})$);
\coordinate (joint-left-heel) at ($ (joint-left-knee) + (270-\kneeAngle-\heelAngle:{2*\Head})$);
\coordinate (joint-right-heel) at ($ (joint-right-knee) + (-90+\kneeAngle+\heelAngle:{2*\Head})$);

\coordinate (dosro-midle-left) at ($(joint-left-hip) + (0,{0.8*\Head})$);
\coordinate (dosro-midle-right) at ($(joint-right-hip) + (0,{0.8*\Head})$);
\coordinate (hips-left) at ($(joint-left-hip) + ({-0.2*\Head},0)$);
\coordinate (hips-right) at ($(joint-right-hip) + ({0.2*\Head},0)$);

\draw[DBlue, fill=#2, thick, line cap=round] (head) circle ({0.5*\Head});

\draw[DBlue, thick, line cap=round] (joint-left-hip) -- (joint-left-knee);
\draw[DBlue, thick, line cap=round] (joint-right-hip) -- (joint-right-knee);

\draw[DBlue, thick, line cap=round] (joint-left-knee) -- (joint-left-heel);
\draw[DBlue, thick, line cap=round] (joint-right-knee) -- (joint-right-heel);

\draw[DBlue, thick, line cap=round] (joint-left-sholder) -- (joint-left-elbow);
\draw[DBlue, thick, line cap=round] (joint-right-sholder) -- (joint-right-elbow);

\draw[DBlue, thick, line cap=round] (joint-left-elbow) -- (joint-left-wrist);
\draw[DBlue, thick, line cap=round] (joint-right-elbow) -- (joint-right-wrist);

\draw[DBlue, thick, line cap=round, fill=#2] (joint-right-sholder) parabola bend (0,{2.7*\Head}) (joint-left-sholder) to[out=-90,in=90]
        (dosro-midle-left) to[out=-90,in=90] (hips-left) parabola bend (0,{-0.1*\Head})
        (hips-right) to[out=90,in=-90] (dosro-midle-right) to[out=90,in=-90] (joint-right-sholder);

\coordinate (dorso-vcenter) at (0,{1.3*\Head});
\node[color=#3, anchor=center, align=center, font=\bfseries] at (dorso-vcenter) {#4};
\end{scope}
}

\NewDocumentCommand{\DrawOpinionPos}{ O{(0,0)} }{
    \DrawPerson[#1]
}
\NewDocumentCommand{\DrawOpinionNeg}{ O{(0,0)} }{
    \DrawPerson[#1][LBlue][black][0]
}

\begin{document}

\begin{tikzpicture}[x=1em, y=1em]

\pgfmathsetmacro{\Height}{ 2.5 }
\pgfmathsetmacro{\Head}{ 0.25*\Height }

\pgfmathsetmacro{\elbowAngle}{ 20 }
\pgfmathsetmacro{\wristAngle}{ -20 }
\pgfmathsetmacro{\kneeAngle}{ 0 }
\pgfmathsetmacro{\heelAngle}{ -20 }


\pgfmathdeclarefunction{SholderRy}{1}{%
    \pgfmathparse{(0.75*sin(#1))*\Head}%
}
\pgfmathdeclarefunction{SholderRx}{1}{%
    \pgfmathparse{0.75*cos(#1)*\Head}%
}
\pgfmathdeclarefunction{SholderLy}{1}{%
    \pgfmathparse{(-0.75*sin(#1))*\Head}%
}
\pgfmathdeclarefunction{SholderLx}{1}{%
    \pgfmathparse{-0.75*cos(#1)*\Head}%
}

\pgfmathdeclarefunction{HipsRy}{1}{%
    \pgfmathparse{0.52*sin(#1)*\Head}%
}
\pgfmathdeclarefunction{HipsRx}{1}{%
    \pgfmathparse{0.52*cos(#1)*\Head}%
}
\pgfmathdeclarefunction{HipsLy}{1}{%
    \pgfmathparse{(-0.52*sin(#1))*\Head}%
}
\pgfmathdeclarefunction{HipsLx}{1}{%
    \pgfmathparse{-0.52*cos(#1)*\Head}%
}


\draw[gray, thick] (0,0) -- (-5.2,0);
\DrawOpinionPos[(-5.2,0)]

\draw[gray, thick] (0,0) -- (0,+5.2);
\DrawOpinionPos[(0,+5.2)]

\draw[gray, thick] (0,0) -- (0,-5.2);
\DrawOpinionPos[(0,-5.2)]

\draw[gray, thick] (0,0) -- (5.2,0);
\DrawOpinionNeg[(5.2,0)]

\DrawOpinionNeg

\end{tikzpicture}

\end{document}
