\NewDocumentCommand{\DrawSpins}{O{A1}}{
    \begin{scope}[shift={(#1-center)}]
    \foreach \i in {-5,0,5}{
        \draw[gray, thick] (\i,-5) -- (\i,-7);
        \draw[gray, thick] (\i,5) -- (\i,7);
        \draw[gray, thick] (\i,-5) -- (\i,0);
        \draw[gray, thick] (\i,5) -- (\i,0);
    }
    \foreach \j in {-5,0,5}{
        \draw[gray, thick] (-5,\j) -- (-7,\j);
        \draw[gray, thick] (5,\j) -- (7,\j);
        \draw[gray, thick] (-5,\j) -- (0,\j);
        \draw[gray, thick] (5,\j) -- (0,\j);
    }
    \end{scope}

    \AgentPlus[(#1-east-agent)]
    \AgentMinus[(#1-west-agent)]
    \AgentMinus[(#1-north-agent)]
    \AgentMinus[(#1-south-agent)]
    \AgentPlus[(#1-northeast-agent)]
    \AgentMinus[(#1-southeast-agent)]
    \AgentPlus[(#1-southwest-agent)]
    \AgentPlus[(#1-northwest-agent)]
}
